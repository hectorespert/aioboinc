320def TIME_TAG_PARSER(x: str) -> datetime | None: 

321 # BOINC sends a time as seconds since the epoch, which name an instant whatever the time zone. 

322 # Without a tzinfo it would be read in the time zone of the host running this client, which 

323 # need not be the one of the host running BOINC, and could not be compared with an aware time. 

324 seconds = float(x) 

325 # BOINC sends 0 for a time that has not been set, such as the next_rpc_time of a project that 

326 # has not asked for one. It is not 1970. 

327 if seconds == 0: 

328 return None 

329 return datetime.fromtimestamp(seconds, UTC)

333 return timedelta(seconds=float(x))

336def ENUM_TAG_PARSER(enum: type[IntEnum]) -> Callable[[str], int]: 

337 def parse(x: str) -> int: 

338 value = int(x) 

339 try: 

340 return enum(value) 

341 except ValueError: 

342 # A value the enum does not have, such as a new reason in a newer BOINC client, is still 

343 # a number, not an error 

344 return value 

345 

346 return parse

393def parse_coprocs(e: ET.Element) -> dict[str, Any]: 

394 # BOINC writes one element for each kind of GPU, with how many there are in count, so a dict by 

395 # the name of the resource keeps them all, and says which kind each is. Any other coprocessor is 

396 # a <coproc> with the name of its resource in <type>. 

397 return {GPU_RESOURCE.get(c.tag) or c.findtext(Tag.TYPE) or c.tag: parse_generic(c) for c in e}

400def parse_generic(e: ET.Element): 

401 if e.tag == Tag.COPROCS: 

402 return parse_coprocs(e) 

403 if e.tag in LIST_TAG: 

404 # Checked first, so that an empty list is [], whether it comes as <results/> or with the 

405 # newline between its tags. 

406 return [parse_generic(cc) for cc in e] 

407 elif len(e) > 0: 

408 # recurse on elements with children 

409 res: dict[str, Any] = {} 

410 for c in e: 

411 if c.tag in BY_RESOURCE_TAG: 

412 value = BY_RESOURCE_TAG[c.tag](c.findtext(Tag.VALUE, "0")) 

413 res.setdefault(c.tag, {})[c.findtext(Tag.NAME, "")] = value 

414 elif c.tag in RESOURCE_NAMES_TAG: 

415 res.setdefault(c.tag, []).append(c.text or "") 

416 elif c.tag in REPEATED_TAG: 

417 res.setdefault(c.tag, []).append(parse_generic(c)) 

418 else: 

419 res[c.tag] = parse_generic(c) 

420 return res 

421 elif e.text is not None: 

422 # set parsed value if available 

423 return TAG_PARSER.get(e.tag, str)(e.text) 

424 else: 

425 # ElementTree reads <tag/> and <tag></tag> the same, so whether an empty element is a flag 

426 # that is set or a text that is empty depends on the tag. 

427 return True if e.tag in BOOL_TAG else ""

455def parse_result(e: ET.Element) -> dict[str, Any]: 

456 # The flags are there, False, when BOINC does not send them 

457 result: dict[str, Any] = dict.fromkeys(RESULT_FLAG_TAG, False) 

458 result.update(parse_generic(e)) 

459 if Tag.ACTIVE_TASK in result: 

460 active_task: dict[str, Any] = dict.fromkeys(ACTIVE_TASK_FLAG_TAG, False) 

461 active_task.update(result[Tag.ACTIVE_TASK]) 

462 result[Tag.ACTIVE_TASK] = active_task 

463 return result

466def parse_project(e: ET.Element) -> "Project": 

467 # The lists of resources, and of links, are there, empty, when BOINC sends none 

468 fields: dict[str, Any] = {tag: [] for tag in RESOURCE_NAMES_TAG}
